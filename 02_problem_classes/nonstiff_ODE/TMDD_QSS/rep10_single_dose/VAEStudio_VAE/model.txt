[PARAMETERS]
Cl
V1
Q
V2
R0
kdeg
kint
KSS

[EQUATIONS]
Ctot = Atot/V1
Dq = Ctot - Rtot - KSS
C = 0.5*(Dq + sqrt(Dq^2 + 4*KSS*Ctot))
Occ = Rtot*C/(KSS + C)
ddt_Atot = -Cl*C - Q*C + Q*A2/V2 - kint*V1*Occ
ddt_A2 = Q*C - Q*A2/V2
ddt_Rtot = kdeg*R0 - kdeg*Rtot - (kint - kdeg)*Occ
Rtot(0) = R0

[DOSE]
; <name>: <data column> -> <state>, kind = bolus|rate[, from = <time>, until = <time>]
Dose: Dose -> Atot, kind = bolus

[OUTPUT]
; <observation in the dataset> = <model variable>
Free = C
Target = Rtot
